rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function fam(fid) { return get(/databases/$(database)/documents/families/$(fid)).data; }
    function isMember(fid) { return signedIn() && request.auth.uid in fam(fid).members; }
    function myRole(fid) { return fam(fid).members[request.auth.uid].role; }
    function isAdmin(fid) { return isMember(fid) && myRole(fid) == 'admin'; }
    function isGuardian(fid) { return isMember(fid) && myRole(fid) in ['admin', 'responsavel']; }
    function isAdult(fid) { return isMember(fid) && myRole(fid) in ['admin', 'responsavel', 'membro']; }
    function visOk(fid, data) { return isGuardian(fid) || data.vis != 'restrito'; }
    function inviteOk(fid, code) {
      let p = /databases/$(database)/documents/families/$(fid)/invites/$(code);
      return exists(p) && get(p).data.active == true && get(p).data.expiresAt > request.time.toMillis() && get(p).data.uses < get(p).data.maxUses;
    }

    match /users/{uid} {
      allow read, write: if signedIn() && request.auth.uid == uid;
    }
    match /families/{fid} {
      allow read: if isMember(fid);
      allow create: if signedIn()
        && request.resource.data.owner == request.auth.uid
        && request.resource.data.members.keys().hasOnly([request.auth.uid])
        && request.resource.data.members[request.auth.uid].role == 'admin';

      // Chefes (admin): mudam tudo, mas o criador continua chefe e só ele passa a criação para outro chefe.
      function ownerOk() {
        let o = request.resource.data.owner;
        return o in request.resource.data.members && request.resource.data.members[o].role == 'admin'
          && (o == resource.data.owner || request.auth.uid == resource.data.owner);
      }
      // Responsáveis: só podem aprovar a entrada de membro adulto ou criança (um por vez).
      function guardianAddsMember() {
        let d = request.resource.data.diff(resource.data);
        let m = request.resource.data.members.diff(resource.data.members);
        return d.affectedKeys().hasOnly(['members']) && m.removedKeys().size() == 0 && m.changedKeys().size() == 0
          && m.addedKeys().size() == 1 && request.resource.data.members[m.addedKeys().toList()[0]].role in ['membro', 'crianca'];
      }
      // Qualquer um (menos o criador) pode sair da família sozinho.
      function leavingSelf() {
        let d = request.resource.data.diff(resource.data);
        let m = request.resource.data.members.diff(resource.data.members);
        return d.affectedKeys().hasOnly(['members']) && m.addedKeys().size() == 0 && m.changedKeys().size() == 0
          && m.removedKeys().hasOnly([request.auth.uid]) && request.auth.uid != resource.data.owner;
      }
      // Chefe não rebaixa nem remove outro chefe sozinho: precisa de pedido (gov/hd_<uid>) aprovado por outro chefe
      // ou vencido (48 h) sem veto. Mudar a si mesmo ou quem não é chefe continua direto.
      function govOk(t) {
        let p = /databases/$(database)/documents/families/$(fid)/gov/$('hd_' + t);
        return exists(p) && (get(p).data.status == 'aprovado' || (get(p).data.status == 'pendente' && get(p).data.execAfterMs < request.time.toMillis()));
      }
      function headsOk() {
        let m = request.resource.data.members.diff(resource.data.members);
        let ch = m.changedKeys().union(m.removedKeys());
        return ch.size() == 0 || (ch.size() == 1 && (resource.data.members[ch.toList()[0]].role != 'admin'
          || ch.toList()[0] == request.auth.uid
          || (ch.toList()[0] in request.resource.data.members && request.resource.data.members[ch.toList()[0]].role == 'admin')
          || govOk(ch.toList()[0])));
      }
      // Acesso de emergência: o contato de emergência libera o próprio acesso ao cofre só depois da espera, se ninguém vetou.
      function emergencySelf() {
        let d = request.resource.data.diff(resource.data);
        let m = request.resource.data.members.diff(resource.data.members);
        let p = /databases/$(database)/documents/families/$(fid)/gov/$('em_' + request.auth.uid);
        return d.affectedKeys().hasOnly(['members']) && m.changedKeys().hasOnly([request.auth.uid]) && m.addedKeys().size() == 0 && m.removedKeys().size() == 0
          && request.resource.data.members[request.auth.uid].diff(resource.data.members[request.auth.uid]).affectedKeys().hasOnly(['emergencyVault'])
          && request.auth.uid in resource.data.policy.emContacts
          && exists(p) && get(p).data.status == 'pendente' && get(p).data.execAfterMs < request.time.toMillis();
      }
      allow update: if (isAdmin(fid) && ownerOk() && headsOk()) || (isGuardian(fid) && guardianAddsMember()) || (isMember(fid) && leavingSelf()) || (isMember(fid) && emergencySelf());
      allow delete: if signedIn() && resource.data.owner == request.auth.uid;

      // Convites: quem tem o código (o segredo) consegue ler aquele convite; só chefes e responsáveis criam e revogam.
      match /invites/{code} {
        allow get: if signedIn();
        allow list, create, update, delete: if isGuardian(fid);
      }
      // Pedido de entrada: só com convite válido (ativo, dentro do prazo e com vagas).
      match /joinRequests/{uid} {
        allow create: if signedIn() && request.auth.uid == uid && request.resource.data.code is string && inviteOk(fid, request.resource.data.code);
        allow update: if isGuardian(fid);
        allow read, delete: if (signedIn() && request.auth.uid == uid) || isGuardian(fid);
      }
      // Histórico de segurança: só se acrescenta; ninguém edita nem apaga.
      match /audit/{id} {
        allow read: if isGuardian(fid);
        allow create: if isMember(fid) && request.resource.data.byUid == request.auth.uid;
      }
      match /vault/{id} {
        allow read: if isGuardian(fid) || (isMember(fid) && fam(fid).members[request.auth.uid].emergencyVault == true);
        allow write: if isGuardian(fid);
      }
      // Pedidos de governança (rebaixar/remover chefe, passar a criação, emergência).
      function govNew(gid) {
        let g = request.resource.data;
        return g.by == request.auth.uid && g.status == 'pendente' && (
          (isAdmin(fid) && g.type in ['rebaixar', 'remover'] && gid == 'hd_' + g.target && g.execAfterMs >= request.time.toMillis() + 47 * 3600000)
          || (isAdmin(fid) && g.type == 'transferir' && request.auth.uid == fam(fid).owner)
          || (isMember(fid) && g.type == 'emergencia' && gid == 'em_' + request.auth.uid && request.auth.uid in fam(fid).policy.emContacts
              && g.execAfterMs >= request.time.toMillis() + fam(fid).policy.emWaitDays * 86400000 - 3600000));
      }
      function govAnswer() {
        let d = request.resource.data.diff(resource.data);
        let s = request.resource.data.status;
        return d.affectedKeys().hasOnly(['status', 'approvals', 'answeredAt', 'doneAt']) && (
          (resource.data.target == request.auth.uid && s in ['aceito', 'recusado', 'vetado'])
          || (resource.data.by == request.auth.uid && s in ['cancelado', 'executado', 'liberado'])
          || (isAdmin(fid) && s == 'aprovado' && resource.data.by != request.auth.uid && resource.data.target != request.auth.uid)
          || (isGuardian(fid) && s == 'vetado' && resource.data.by != request.auth.uid)
          || (isAdmin(fid) && s == 'executado'));
      }
      match /gov/{gid} {
        allow read: if isMember(fid);
        allow create: if govNew(gid);
        allow update: if govAnswer() || (resource.data.status in ['vetado', 'cancelado', 'executado', 'recusado', 'liberado'] && govNew(gid));
      }
      // Aparelhos conectados: cada um registra o seu; chefe desconecta; o aparelho não se "religa" sozinho.
      match /devices/{did} {
        allow read: if isGuardian(fid) || (isMember(fid) && resource.data.uid == request.auth.uid);
        allow create: if isMember(fid) && request.resource.data.uid == request.auth.uid && !('revoked' in request.resource.data);
        allow update: if isAdmin(fid) || (isMember(fid) && resource.data.uid == request.auth.uid
          && !request.resource.data.diff(resource.data).affectedKeys().hasAny(['revoked', 'revokedBy', 'revokedAt', 'uid']))
          || (isMember(fid) && resource.data.uid == request.auth.uid && request.resource.data.revoked == true);
        allow delete: if isAdmin(fid);
      }
      // Duas casas: registro de combinados só recebe itens novos; ninguém edita nem apaga.
      match /comsg/{id} {
        allow read: if isMember(fid);
        allow create: if isAdult(fid) && request.resource.data.at is string;
      }
      match /requests/{id} {
        allow read: if isMember(fid) && visOk(fid, resource.data);
        allow create: if isMember(fid) && request.resource.data.byUid == request.auth.uid;
        allow update: if isGuardian(fid) || (isMember(fid) && resource.data.byUid == request.auth.uid)
          || (isAdult(fid) && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status', 'boughtAt', 'updatedAt', 'updatedBy']));
        allow delete: if isGuardian(fid) || (isMember(fid) && resource.data.byUid == request.auth.uid);
      }
      match /files/{fileId}/chunks/{n} {
        allow read: if isMember(fid) && visOk(fid, resource.data);
        allow write: if isMember(fid);
      }
      match /{col}/{id} {
        allow read: if !(col in ['vault', 'joinRequests', 'invites', 'audit', 'gov', 'devices', 'comsg']) && isMember(fid) && visOk(fid, resource.data);
        allow create, update: if !(col in ['vault', 'requests', 'joinRequests', 'invites', 'audit', 'gov', 'devices', 'comsg'])
          && (isAdult(fid) || (isMember(fid) && col in ['tasks', 'shop', 'items', 'msgs', 'reminders', 'files', 'checkins']));
        allow delete: if !(col in ['vault', 'requests', 'joinRequests', 'invites', 'audit', 'gov', 'devices', 'comsg']) && isAdult(fid);
      }
    }
  }
}
